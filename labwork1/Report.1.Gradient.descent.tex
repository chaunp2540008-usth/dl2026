\begin{abstract}
This lab work reports a gradient descent experiment on the function $f(x)=x^{2}$. The study evaluates the convergence behaviour across a range of learning rates.
\end{abstract}

\section{Implementation}
The experiment uses initial value $x_{0}=10$ and convergence threshold $\epsilon=10^{-6}$ on $f(x)$.

\begin{verbatim}
def grad(fod, x, lr):
    return x - lr * fod(x)

def f(x):
    return x * x

def parabolic_fod(x):
    return 2 * x

lr = 0.1
epsilon = 1e-6
x = 10
iteration = 0
while f(x) > epsilon:
    x = grad(parabolic_fod, x, lr)
    iteration += 1
    print(iteration, x, f(x))
\end{verbatim}
